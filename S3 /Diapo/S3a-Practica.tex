\frame{\frametitle{Pregunta 2: solución}

\begin{itemize}	
%\pause

\item La tasa de Kaplan-Meier es definida como:

$$
\hat{\theta }(T_{k})=\frac{h_{k}}{n_{k}} 
$$

%\pause
\item Donde $h_{k}$ es el numero de \textbf{falla} del estado en el tiempo $k$  

%\pause
\item Luego, $n_{k}$ es el \tetxbf{conjunto de riesgo} en el tiempo $k$ (\texttt{conjunto de personas que se mantiene en el analisis})

%\pause
\item El conjunto de riesgo comprende todas aquellas unidades en riesgo de \textbf{salir del estado (o fallar)} en el momento relevante.

\end{itemize}	

}

Interprete los coeficientes estimados para la \texttt{edad}(age) , \texttt{raza negros} (black), sentencias de condena previos (\texttt{priors}). \end{itemize}

\frame{\frametitle{Pregunta 3: solución}

\begin{itemize}	
\item El modelo \textbf{exponencial de hazard} puede ser expresada como:

$$
\theta_{it}=exp( \beta_{0}+\beta_{1}age_{i}+\beta_{2}married_{i}+\beta_{3}black_{i} + \cdots 
$$

$$
\cdots + \beta_{4} drugs_{i} +\beta_{5}alcohol_{i}+\beta_{6}priors_{i}+\beta_{7}rules_{i}+\beta_{8}tserved_{i} ) 
$$

\item Podriamos observar que ninguna de las variables explicativas \textbf{cambia respecto al tiempo}

\item Todas las covariables exiben realizaciones para individuos previo a la fecha de su liberación para que no cambie mientras se acerca al tiempo de falla. 
\end{itemize}	

}

\frame{\frametitle{Pregunta 3: solución}

\begin{itemize}	
\item El logaritmo del modelo \textbf{exponencial de hazard} puede ser expresada como:

$$
log_{e} (\theta_{it})= \beta_{0}+\beta_{1}\textcolor{blue}{age_{i}}+\beta_{2}\textcolor{red}{married_{i}}+\beta_{3}\textcolor{red}{black_{i}} + \cdots 
$$

$$
\cdots + \beta_{4} \textcolor{red}{drugs_{i}} +\beta_{5}\textcolor{red}{alcohol_{i}}+\beta_{6}\textcolor{blue}{priors_{i}}+\beta_{7}\textcolor{blue}{rules_{i}}+\beta_{8}\textcolor{blue}{tserved_{i}} 
$$

\item \textcolor{red}{Para la  $k^{th}$ variable binaria }: 

\begin{center}
$(exp (\hat{\beta }_{k} )-1) $ x $100$
\end{center}

\item \textcolor{blue}{Para la  $k^{th}$ variable continua} : 

$$ \frac{ \partial log( \hat{\beta }_{k} )}{\partial X_{k}} =\hat{\beta }_{k}  $$

\end{itemize}

\end{center}	

\item \small Note que $H_{o}: ln(\alpha )=0$ es equivalente a $\alpha =1$

\item \small La pregunta ahora es como derivar la varianza muestral en este caso?

\item \small Respuesta : podemos usar el \tetxbf{metodo delta}
\end{itemize}

\frame{\frametitle{Pregunta 4.2: solución}

\begin{itemize}	
\item \small Si $\bigtriangleup = ln(\alpha )$ entonces:

$$
\frac{\partial \bigtriangleup }{\partial \alpha }=\frac{1}{\alpha }
$$

\item \small Esta expresión para estimar la varianza muestral  estará dado por:

$$
Var (\hat{\bigtriangleup } ) = \left( \frac{\partial \bigtriangleup }{\partial \alpha } \right)^{2} Var(\hat{\alpha } ) 
$$

$$
Var (\hat{\bigtriangleup } ) = \left( \frac{1}{ \alpha } \right)^{2} Var(\hat{\alpha } ) 
$$

\item \small Insertamos los valores empiricos estimados y obtenemos:

$$
Var (\hat{\bigtriangleup } ) =\left( \frac{1}{ 0.8042664} \right)^{2} (0.000979564804) 
$$

$$
Var (\hat{\bigtriangleup } ) =0.001514374616
$$


\end{itemize}	

}

$$
